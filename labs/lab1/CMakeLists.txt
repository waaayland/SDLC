cmake_minimum_required(VERSION 3.21)

project(PasswordGenerator
    VERSION 1.0.0
    DESCRIPTION "Лабораторная работа №1 — генератор паролей (MVC с активной моделью)"
    LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Исходники содержат строковые литералы в UTF-8
if (MSVC)
    add_compile_options(/utf-8)
endif()

find_package(Qt6 REQUIRED COMPONENTS Widgets)

qt_standard_project_setup()

# ---------------------------------------------------------------------------
# Кириллица в путях. На Windows с системной кодировкой CP1251 инструменты
# сборки по-разному переживают не-ASCII пути:
#
#   * mingw32-make  — не работает совсем: не видит ни исходники, ни
#                     AutogenInfo.json, если в пути есть кириллица;
#   * ninja         — работает корректно;
#   * moc / uic     — читают из кириллических путей, но не могут в них ПИСАТЬ.
#
# Отсюда два правила ниже: требуем генератор Ninja и уводим автогенерируемые
# файлы (moc_*, ui_*) в каталог с латинским путём.
# ---------------------------------------------------------------------------
string(REGEX MATCH "[^ -~]" _NON_ASCII_IN_BINARY_DIR "${CMAKE_BINARY_DIR}")
string(REGEX MATCH "[^ -~]" _NON_ASCII_IN_SOURCE_DIR "${CMAKE_SOURCE_DIR}")

if ((_NON_ASCII_IN_BINARY_DIR OR _NON_ASCII_IN_SOURCE_DIR)
    AND CMAKE_GENERATOR MATCHES "Makefiles")
    # ВНИМАНИЕ: текст сообщения намеренно на латинице. CMake выводит его
    # в UTF-8, а Qt Creator читает вывод в системной кодировке (CP1251),
    # поэтому кириллица здесь превратилась бы в нечитаемые символы.
    message(FATAL_ERROR
        "\n"
        "=== NON-ASCII PATH + \"${CMAKE_GENERATOR}\" GENERATOR ===\n"
        "\n"
        "Put' k proektu ili k katalogu sborki soderzhit kirillicu:\n"
        "  source: ${CMAKE_SOURCE_DIR}\n"
        "  build : ${CMAKE_BINARY_DIR}\n"
        "\n"
        "mingw32-make ne umeet rabotat' s takimi putyami: on ne naydet\n"
        "ni ishodniki, ni AutogenInfo.json. Nuzhen generator Ninja.\n"
        "\n"
        "Qt Creator:\n"
        "  Instrumenty -> Parametry -> Komplekty -> vybrat' komplekt\n"
        "  \"Desktop Qt 6.x MinGW 64-bit\" -> pole \"CMake generator\"\n"
        "  -> Izmenit'... -> Generator: Ninja -> OK -> PRIMENIT' -> OK.\n"
        "  Zatem: Sborka -> Ochistit' konfiguraciyu -> Zapustit' CMake.\n"
        "\n"
        "Komandnaya stroka:\n"
        "  cmake -S . -B <katalog> -G Ninja\n"
        "  (ili prosto zapustite build.bat iz papki proekta)\n"
        "\n"
        "Alternativa: perenesti proekt v put' bez kirillicy, naprimer\n"
        "  C:/dev/01_LAB\n")
endif()

if (_NON_ASCII_IN_BINARY_DIR)
    string(MD5 _PROJECT_HASH "${CMAKE_BINARY_DIR}")
    string(SUBSTRING "${_PROJECT_HASH}" 0 8 _PROJECT_HASH)

    if (DEFINED ENV{LOCALAPPDATA})
        set(_AUTOGEN_ROOT "$ENV{LOCALAPPDATA}/QtAutogen")
    elseif (DEFINED ENV{TMPDIR})
        set(_AUTOGEN_ROOT "$ENV{TMPDIR}/QtAutogen")
    else()
        set(_AUTOGEN_ROOT "/tmp/QtAutogen")
    endif()

    set(_AUTOGEN_DIR "${_AUTOGEN_ROOT}/${PROJECT_NAME}-${_PROJECT_HASH}")

    string(REGEX MATCH "[^ -~]" _NON_ASCII_IN_AUTOGEN_DIR "${_AUTOGEN_DIR}")
    if (_NON_ASCII_IN_AUTOGEN_DIR)
        message(FATAL_ERROR
            "Каталог сборки содержит кириллицу, и запасной путь для\n"
            "автогенерируемых файлов — тоже:\n"
            "  ${_AUTOGEN_DIR}\n"
            "Утилита moc из Qt не умеет создавать в таких каталогах файлы.\n"
            "Задайте каталог сборки в пути без кириллицы, например:\n"
            "  cmake -S . -B C:/QtBuilds/${PROJECT_NAME}\n"
            "В Qt Creator: Проекты (Ctrl+5) -> Настройки сборки -> Каталог сборки.")
    endif()

    file(MAKE_DIRECTORY "${_AUTOGEN_DIR}")

    # Латиница — по той же причине, что и в сообщении выше.
    message(STATUS "Build dir contains non-ASCII characters; moc/uic cannot write there.")
    message(STATUS "Autogenerated files redirected to: ${_AUTOGEN_DIR}")
endif()

# Формы Qt Designer лежат рядом с классами представления
set(CMAKE_AUTOUIC_SEARCH_PATHS ${CMAKE_CURRENT_SOURCE_DIR}/src/view)

qt_add_executable(PasswordGenerator WIN32
    src/main.cpp

    # --- Model ---
    src/model/passwordsettings.h
    src/model/rawpasswordinput.h
    src/model/imodelobserver.h
    src/model/passwordgenerator.h
    src/model/passwordgenerator.cpp
    src/model/passwordmodel.h
    src/model/passwordmodel.cpp

    # --- Controller ---
    src/controller/passwordcontroller.h
    src/controller/passwordcontroller.cpp

    # --- View ---
    src/view/mainwindow.h
    src/view/mainwindow.cpp
    src/view/mainwindow.ui
    src/view/inputdialog.h
    src/view/inputdialog.cpp
    src/view/inputdialog.ui
)

# Перенаправление автогенерируемых файлов (см. проверку пути выше).
# AUTOGEN_BUILD_DIR — свойство цели, поэтому задаётся после её создания.
if (_NON_ASCII_IN_BINARY_DIR)
    set_target_properties(PasswordGenerator PROPERTIES
        AUTOGEN_BUILD_DIR "${_AUTOGEN_DIR}")
endif()

target_include_directories(PasswordGenerator PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)

target_link_libraries(PasswordGenerator PRIVATE Qt6::Widgets)

install(TARGETS PasswordGenerator
    RUNTIME DESTINATION bin
    BUNDLE  DESTINATION .)
